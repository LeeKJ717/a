| Python(math와 numpy만 쓴 직접 작성 함수) | 이 장의 계산 재현(ch10-calc-v1.py, ch10-tables-v1.py) | 이 책 | 표준 라이브러리와 numpy |

**이 장의 계산을 직접 해 보는 코드.** Brune 포획 효율, 구간 수지, CIA, εNd, 혼합 비율, 하버사인, 시료 간격과 나이키스트 주기이다. 숫자는 10.10절의 시연값과 같다.

```python
import math
def brune_te(V, Q): return 100 * (1 - 0.05 / math.sqrt(V / Q))
print(round(brune_te(2, 100), 2), round(brune_te(8, 100), 2))      # 64.64 82.32
def budget(inflows, outflow): return sum(inflows) - outflow
print(budget([156, 35, 23, 26], 197))                               # 43
def cia(al, ca, na, k, p):                                           # 몰 값을 넣는다
    cs = min(ca - 10 / 3 * p, na)
    return 100 * al / (al + cs + na + k)
print(round(cia(12.37 / 101.961, 12.82 / 56.077, 1.61 / 61.979, 2.52 / 94.196, 0.24 / 141.945), 2))   # 60.65
def eps_nd(r, chur=0.512638): return (r / chur - 1) * 1e4
print(round(eps_nd(0.512092), 2))                                    # -10.65
def mix_fraction(x, y, z): return (z - x) / (y - x)
print(round(mix_fraction(-11.70, -15.18, -13.30), 2))                # 0.46
def haversine(lo1, la1, lo2, la2, R=6371.0):
    p1, p2 = math.radians(la1), math.radians(la2)
    a = math.sin((p2 - p1) / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin(math.radians(lo2 - lo1) / 2) ** 2
    return 2 * R * math.asin(math.sqrt(a))
print(round(haversine(0, 0, 1, 0), 3))                               # 111.195
def sample_interval_years(cm, rate_cm_per_a): return cm / rate_cm_per_a
print(round(sample_interval_years(1, 0.92), 2), round(2 * sample_interval_years(1, 0.92), 2))   # 1.09 2.17
```
